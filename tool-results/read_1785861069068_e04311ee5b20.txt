   135→/* ------------------------------------------------------------------ */
   136→/*  SECTION WRAPPER WITH FADE-IN                                      */
   137→/* ------------------------------------------------------------------ */
   138→function FadeSection({
   139→  children,
   140→  className = "",
   141→  delay = 0,
   142→}: {
   143→  children: React.ReactNode;
   144→  className?: string;
   145→  delay?: number;
   146→}) {
   147→  const ref = useRef(null);
   148→  const isInView = useInView(ref, { once: true, margin: "-60px" });
   149→
   150→  return (
   151→    <motion.section
   152→      ref={ref}
   153→      initial={{ opacity: 0, y: 40 }}
   154→      animate={isInView ? { opacity: 1, y: 0 } : {}}
   155→      transition={{ duration: 0.7, delay, ease: "easeOut" }}
   156→      className={className}
   157→    >
   158→      {children}
   159→    </motion.section>
   160→  );
   161→}
