   138→   138→function FadeSection({
   139→   139→  children,
   140→   140→  className = "",
   141→   141→  delay = 0,
   142→   142→}: {
   143→   143→  children: React.ReactNode;
   144→   144→  className?: string;
   145→   145→  delay?: number;
   146→   146→}) {
   147→   147→  const ref = useRef(null);
   148→   148→  const isInView = useInView(ref, { once: true, margin: "-60px" });
   149→   149→
   150→   150→  return (
   151→   151→    <motion.section
   152→   152→      ref={ref}
   153→   153→      initial={{ opacity: 0, y: 40 }}
   154→   154→      animate={isInView ? { opacity: 1, y: 0 } : {}}
   155→   155→      transition={{ duration: 0.7, delay, ease: "easeOut" }}
   156→   156→      className={className}
   157→   157→    >
   158→   158→      {children}
   159→   159→    </motion.section>
   160→   160→  );
   161→   161→}
